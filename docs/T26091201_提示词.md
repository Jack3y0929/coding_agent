### `hello-dev/script.js`
```js
'use strict';

var TARGET_TEXT = 'DEV 工作流运行成功';
var IDS = { greeting: 'greeting', button: 'action-btn' };

function findElements() {
  return {
    greeting: document.getElementById(IDS.greeting),
    button: document.getElementById(IDS.button)
  };
}

function renderSuccess(greeting) {
  greeting.textContent = TARGET_TEXT;   // 用 textContent，不用 innerHTML
  greeting.setAttribute('data-state', 'success');
}

function bindEvents(greeting, button) {
  button.addEventListener('click', function () {
    renderSuccess(greeting);            // 幂等：重复点击结果一致
  });
}

function init() {
  var els = findElements();
  if (!els.greeting) {
    console.error('[hello-dev] 缺少文案元素 #' + IDS.greeting);
    return;
  }
  if (!els.button) {
    console.error('[hello-dev] 缺少按钮元素 #' + IDS.button);
    return;
  }
  bindEvents(els.greeting, els.button);
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', init);
} else {
  init();
}
```

## 4. 数据流设计（单向，无状态存储）

```
浏览器加载 index.html → 解析 DOM(#greeting/#action-btn) → style.css 应用样式
→ script.js 就绪(readyState 双分支) → DOMContentLoaded → init()
→ 查找元素(缺失即 console.error + return) → bindEvents 注册 click
→ [用户点击] → renderSuccess() → textContent = "DEV 工作流运行成功"
                                    + data-state="success" → CSS 派生视觉态
```
无后端、无持久化、无路由；点击结果幂等（重复点击同值）。

## 5. 异常处理策略

| 场景 | 处置 |
|---|---|
| `#greeting` / `#action-btn` 缺失（脚本先于 DOM、id 拼错） | `readyState` 双分支兜底 + null 判空 → `console.error` 后安全 return，不抛异常 |
| file:// 下脚本不执行 | 禁用 module / fetch / import；普通 `<script>` + 相对路径 |
| 中文文案乱码 | `<meta charset="UTF-8">` + 写入 UTF-8（无 BOM） |
| XSS / 转义歧义 | 全程 `textContent`，不用 `innerHTML` |
| 无 Node | 降级为语法自查（见下 R1） |
