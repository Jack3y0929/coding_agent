### 3. `hello-dev/script.js`（新建）

```js
(function () {
  'use strict';

  const TEXT = {
    initial: 'Hello DEV',
    success: 'DEV 工作流运行成功'
  };

  const IDS = {
    greeting: 'greeting',
    runBtn: 'run-btn'
  };

  /** 解析必需元素；任一缺失则告警并返回 null，不抛异常 */
  function resolveElements() {
    const greeting = document.getElementById(IDS.greeting);
    const runBtn = document.getElementById(IDS.runBtn);
    const missing = [];

    if (!greeting) { missing.push('#' + IDS.greeting); }
    if (!runBtn) { missing.push('#' + IDS.runBtn); }

    if (missing.length > 0) {
      console.warn('[hello-dev] 缺少必要元素: ' + missing.join(', '));
      return null;
    }
    return { greeting: greeting, runBtn: runBtn };
  }

  /** 归一初始文案：即使 HTML 静态文案被改动，重开也一致（幂等） */
  function renderInitial(greeting) {
    greeting.textContent = TEXT.initial;
    greeting.removeAttribute('data-state');
  }

  /** 终态渲染：textContent 为替换语义，不追加节点 */
  function renderSuccess(greeting, runBtn) {
    greeting.textContent = TEXT.success;
    greeting.setAttribute('data-state', 'success');
    runBtn.disabled = true;
  }

  function init() {
    const els = resolveElements();
    if (!els) { return; }

    renderInitial(els.greeting);
    els.runBtn.addEventListener('click', function () {
      renderSuccess(els.greeting, els.runBtn);
    });
  }

  init();
})();
```

## 实现说明（供审查）

**数据流（单向，无状态存储）**

```
点击 #run-btn
  → click 监听器 → renderSuccess(greeting, runBtn)
      → 读内存常量 TEXT.success
      → 写 DOM 文本节点（textContent，替换整个文本子节点）
      → 写 DOM 属性 data-state="success" → CSS 派生视觉
      → runBtn.disabled = true（锁定终态）
```

**关键设计决策**

1. **`file://` 硬指标优先**：脚本置于 `</body>` 前且不加 `type="module"` / `integrity` / CDN；全部相对路径。这是"双击可打开"唯一易踩翻的点，已在 HTML 内留注释固化理由。
2. **幂等三重保障**（验收 4）：`textContent` 赋值是替换非追加；常量字符串与节点数量无关；终态 `disabled` 阻断后续事件——即便不禁用，重复赋值结果亦相同。
3. **初始渲染双保险**（验收 3）：`<h1>` 内直接写死 `Hello DEV`，JS 失败/未执行时仍可见；`renderInitial()` 再归一一次，防止 HTML 与 JS 字面量漂移导致的不一致。
4. **文案集中**：`TEXT` + `IDS` 两组常量，避免 id 字符串与文案在多处散落。
5. **渐进增强**：功能链路不依赖 CSS，`style.css` 加载失败时按钮与文案替换仍可用。
6. **异常策略**：元素缺失 → `resolveElements()` 返回 `null`，`init()` 提前 `return`，仅 `console.warn`，**不抛未捕获异常**；正常路径零 `warn`/零请求 → 满足验收 2「控制台无 404、无报错」。
7. 规范中「Python 类型注解」「async IO」两条不适用于本任务（纯浏览器端 JS，无 IO）；已以 JSDoc 式注释标注函数意图代替。

- 未使用事件委托/`data-*` 驱动文案，超出最小链路需要。
