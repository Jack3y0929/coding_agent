```python
from typing import TypedDict, Literal
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver

# ---- State 定义 ----
class DevWorkflowState(TypedDict):
    # 当前阶段
    stage: str                         # clarify/analyze/develop_plan/develop_build/review/fix/human_accept
    # 各阶段正式产出（节点间唯一传递的内容）
    clarification_doc: str | None      # 澄清文档
    requirement_doc: str | None        # 需求概括文档
    plan: str | None                   # 开发方案（Plan阶段产出）

    fix_attempt: int                   # 修复尝试次数
    token_used: int                    # 已消耗token
    status: str                        # running/paused/done/failed

# ---- 节点函数骨架 ----

    # 2. 调用 `is_clarification_complete()` 判断信息是否足够
    # 3. 够了就产出 clarification_doc，不够就继续提问
    ...

async def node_analyze(state: DevWorkflowState) -> dict:

    messages = [

    messages = [
        {"role": "system", "content": DEVELOPER_SYSTEM_PROMPT + f"\n当前处于Build模式。Plan方案：\n{state['plan']}"},
        {"role": "user", "content": f"需求概括文档：\n{state['requirement_doc']}\n\n请按Plan方案实施。"}

    review_angle = "逻辑+边界" if state["review_round"] == 1 else "规范+整体合理性"
    messages = [

        {"role": "user", "content": f"需求概括文档：\n{state['requirement_doc']}\n\n" +
                                     f"代码变更：\n{json.dumps(state['code_changes'])}\n\n" +
                                     f"实现说明：\n{state['implementation_note']}\n\n" +
                                     f"RAG注入的规范+错误模式：\n{rag_retrieve_specs_and_errors(state['requirement_doc'])}"}
    ]
    review = await call_deepseek(messages)
    return {"review_report": review["report"],
            "review_passed": review["passed"],
            "review_round": state["review_round"] + 1}

async def node_fix(state: DevWorkflowState) -> dict:
    """开发者修复节点：根据审查报告修改代码"""
    messages = [
        {"role": "system", "content": DEVELOPER_SYSTEM_PROMPT},
        {"role": "user", "content": f"审查报告：\n{state['review_report']}\n\n" +
                                     f"当前代码变更：\n{json.dumps(state['code_changes'])}\n\n" +
                                     f"请逐一修复审查问题。"}
    ]
    result = await call_deepseek(messages, tools=BUILD_MODE_TOOLS)
    return {"code_changes": result["code_changes"],
            "fix_attempt": state["fix_attempt"] + 1}

async def node_human_accept(state: DevWorkflowState) -> dict:
    """人工验收节点：暂停，等待人在Web UI点击通过/不通过"""
    # LangGraph interrupt: 暂停执行，等人通过API resume
    return {"stage": "human_accept", "status": "paused"}

# ---- 条件路由 ----
def route_after_review(state: DevWorkflowState) -> Literal["fix", "human_accept"]:
    if state["review_passed"]:
        return "human_accept"
    if state["fix_attempt"] >= 3:
        return "human_accept"  # 最多修复3次，无论如何给人验收
    return "fix"

def route_after_human(state: DevWorkflowState) -> Literal["fix", "output"]:
    """由Web UI的resume API传入通过/不通过"""
    # human_decision 由外部 resume 时注入
    if state.get("human_decision") == "approved":
        return "output"
    return "fix"

# ---- 构建状态图 ----
def build_dev_workflow(checkpointer: SqliteSaver) -> StateGraph:
    graph = StateGraph(DevWorkflowState)

    graph.add_node("input_gate", node_input_gate)
    graph.add_node("analyze", node_analyze)
    graph.add_node("develop_plan", node_develop_plan)
    graph.add_node("develop_build", node_develop_build)
    graph.add_node("review", node_review)
    graph.add_node("fix", node_fix)
    graph.add_node("human_accept", node_human_accept)
    graph.add_node("output", node_output_summary)

    graph.set_entry_point("input_gate")

    graph.add_edge("input_gate", "analyze")
    graph.add_edge("analyze", "develop_plan")
    graph.add_edge("develop_plan", "develop_build")
    graph.add_edge("develop_build", "review")

    graph.add_conditional_edges("review", route_after_review, {
        "fix": "fix",
        "human_accept": "human_accept"
    })
    graph.add_edge("fix", "review")  # 修复后重新审查

    graph.add_conditional_edges("human_accept", route_after_human, {
        "fix": "fix",
        "output": "output"
    })
    graph.add_edge("output", END)

    return graph.compile(checkpointer=checkpointer)
```

## 2. Token预算管理器

### budget.py

```python
import tiktoken
from typing import Any

class TokenBudget:
    """Token预算管理器：每次AI调用前检查，调用后累加"""

    DEEPSEEK_INPUT_PRICE = 2.0   # 元/百万token
    DEEPSEEK_OUTPUT_PRICE = 8.0  # 元/百万token
    MAX_COST = 5.0               # 硬上限（元）

    def __init__(self, max_cost: float = 5.0):
        self.max_cost = max_cost
        self.total_input_tokens: int = 0
        self.total_output_tokens: int = 0
        self.interrupted: bool = False
        self._encoder = tiktoken.get_encoding("cl100k_base")

    def count_tokens(self, messages: list[dict]) -> int:
        """估算消息列表的token数"""
        total = 0
        for msg in messages:
            content = msg.get("content", "") or ""
            total += len(self._encoder.encode(content))
            total += 4  # 每条消息的格式开销
        return total

    def before_call(self, messages: list[dict]) -> int:
        """调用前检查预算，返回预估输入token数。超限抛异常"""
        input_tokens = self.count_tokens(messages)
        estimated_cost = self._calc_cost(
            self.total_input_tokens + input_tokens,
            self.total_output_tokens
        )
        if estimated_cost >= self.max_cost:
            self.interrupted = True
            raise BudgetExceededError(
                f"预算超限：已用{self.total_input_tokens + self.total_output_tokens}token，"
                f"本次需{input_tokens}token，预估费用{estimated_cost:.2f}元 > {self.max_cost}元上限"
            )
        return input_tokens

    def after_call(self, input_tokens: int, output_tokens: int):
        """调用后累加token"""
        self.total_input_tokens += input_tokens
        self.total_output_tokens += output_tokens

    def get_status(self) -> dict[str, Any]:
        """获取当前预算状态（用于Web UI进度面板）"""
        cost = self._calc_cost(self.total_input_tokens, self.total_output_tokens)
        return {
            "total_input_tokens": self.total_input_tokens,
            "total_output_tokens": self.total_output_tokens,
            "total_tokens": self.total_input_tokens + self.total_output_tokens,
            "estimated_cost_yuan": round(cost, 2),
            "max_cost_yuan": self.max_cost,
            "usage_percent": round(cost / self.max_cost * 100, 1),
            "interrupted": self.interrupted,
        }

    def _calc_cost(self, input_tokens: int, output_tokens: int) -> float:
        return (input_tokens / 1_000_000 * self.DEEPSEEK_INPUT_PRICE +
                output_tokens / 1_000_000 * self.DEEPSEEK_OUTPUT_PRICE)


class BudgetExceededError(Exception):
    """预算超限异常"""
    pass
```

## 3. RAG检索器 — 混合检索 + 余弦相似度

### rag/retriever.py

```python
import numpy as np
import sqlite3
from typing import Any

# ---- 余弦相似度（自建，约20行，零外部向量库依赖） ----

def cosine_similarity(vec_a: np.ndarray, vec_b: np.ndarray) -> float:
    """计算两个向量的余弦相似度"""
    dot_product = np.dot(vec_a, vec_b)
    norm_a = np.linalg.norm(vec_a)
    norm_b = np.linalg.norm(vec_b)
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return float(dot_product / (norm_a * norm_b))

# ---- RRF 融合排序 ----

def rrf_fusion(
    keyword_ranked: list[dict],
    semantic_ranked: list[dict],
    k: int = 60
) -> list[dict]:
    """
    Reciprocal Rank Fusion: 融合关键词和语义两路检索结果
    k=60 是常用常数
    """
    scores: dict[int, float] = {}

    for rank, item in enumerate(keyword_ranked):
        scores[item["id"]] = scores.get(item["id"], 0) + 1 / (k + rank + 1)

    for rank, item in enumerate(semantic_ranked):
        scores[item["id"]] = scores.get(item["id"], 0) + 1 / (k + rank + 1)

    # 按融合分数降序排列
    merged = sorted(scores.items(), key=lambda x: x[1], reverse=True)

    # 重建结果列表（保留原始item数据）
    id_to_item = {item["id"]: item for item in keyword_ranked + semantic_ranked}
    return [id_to_item[item_id] for item_id, _ in merged]

# ---- 混合检索入口 ----

async def hybrid_retrieve(
    query: str,
    db_path: str,
    embed_func,
    top_k: int = 5,
    keyword_weight: float = 0.5
) -> list[dict[str, Any]]:
    """
    混合检索：关键词搜索 + 语义搜索 → RRF融合 → 返回top_k条

    Args:
        query: 查询文本
        db_path: SQLite数据库路径
        embed_func: 向量化函数（接收文本，返回numpy数组）
        top_k: 返回结果数
        keyword_weight: 关键词检索权重（0-1），剩余为语义权重
    """
    # 第1路：关键词检索（SQL LIKE）
    keyword_results = _keyword_search(query, db_path, top_k=top_k * 2)

    # 第2路：语义检索（向量余弦相似度）
    query_vec = embed_func(query)
    semantic_results = _semantic_search(query_vec, db_path, top_k=top_k * 2)

    # RRF融合
    fused = rrf_fusion(keyword_results, semantic_results)

    return fused[:top_k]


def _keyword_search(query: str, db_path: str, top_k: int) -> list[dict]:
    """关键词检索：SQL LIKE模糊匹配"""
    keywords = query.split()
    # 构建多关键词OR条件
    conditions = " OR ".join([f"content LIKE '%{kw}%'" for kw in keywords])

    conn = sqlite3.connect(db_path)
    rows = conn.execute(
        f"SELECT id, content, source_type, source_path FROM rag_embeddings "
        f"WHERE {conditions} LIMIT {top_k}"
    ).fetchall()
    conn.close()

    return [
        {"id": r[0], "content": r[1], "source_type": r[2], "source_path": r[3]}
        for r in rows
    ]


def _semantic_search(query_vec: np.ndarray, db_path: str, top_k: int) -> list[dict]:
    """语义检索：加载所有向量，计算余弦相似度，取top_k"""
    conn = sqlite3.connect(db_path)
    rows = conn.execute(
        "SELECT id, content, embedding, source_type, source_path FROM rag_embeddings"
    ).fetchall()
    conn.close()

    scored = []
    for row in rows:
        stored_vec = np.frombuffer(row[2], dtype=np.float32)
        sim = cosine_similarity(query_vec, stored_vec)
        scored.append({
            "id": row[0], "content": row[1], "source_type": row[3],
            "source_path": row[4], "score": sim
        })

    scored.sort(key=lambda x: x["score"], reverse=True)
    return scored[:top_k]
```

## 5. Shell命令白名单校验

### tools/shell_tools.py

```python
import asyncio
import logging

logger = logging.getLogger(__name__)

# 白名单：仅允许编译、检查、读取类命令
SHELL_WHITELIST = [
    "cargo check",
    "cargo build",
    "cargo test",
    "wasm-pack build",
    "npm run build",
    "npm run lint",
    "npm run typecheck",
    "npx tsc --noEmit",
    "pytest",
    "python -m pytest",
    "git diff",
    "git status",
    "git log --oneline",
]

def validate_command(command: str) -> bool:
    """检查命令是否在白名单内（前缀匹配）"""
    cmd = command.strip()
    for allowed in SHELL_WHITELIST:
        if cmd.startswith(allowed):
            return True
    logger.warning(f"Shell命令被拒绝（不在白名单）: {cmd}")
    return False


async def execute_shell(command: str, timeout: int = 60, cwd: str | None = None) -> str:
    """执行白名单内的shell命令，返回stdout+stderr"""
    if not validate_command(command):
        return f"拒绝执行：命令 '{command}' 不在白名单内。允许的命令: {SHELL_WHITELIST}"

    try:
        process = await asyncio.create_subprocess_shell(
            command,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
            cwd=cwd,
        )
        stdout, stderr = await asyncio.wait_for(
            process.communicate(), timeout=timeout
        )
        output = stdout.decode("utf-8", errors="replace")
        if stderr:
            output += "\n[stderr]\n" + stderr.decode("utf-8", errors="replace")
        output += f"\n[exit_code: {process.returncode}]"
        return output
    except asyncio.TimeoutError:
        return f"命令超时（>{timeout}秒）: {command}"
    except Exception as e:
        return f"命令执行异常: {e}"
```

## 6. DeepSeek API 调用封装

### api.py

```python
from openai import AsyncOpenAI

async def call_deepseek(
    messages: list[dict],
    tools: list[dict] | None = None,
    model: str = "deepseek-chat",
    api_key: str = "",
    budget: "TokenBudget | None" = None,
    max_tool_rounds: int = 10,
) -> str:
    """
    调用DeepSeek API，支持FC工具循环。
    返回最终文本结果。
    """
    client = AsyncOpenAI(api_key=api_key, base_url="https://api.deepseek.com")

    # 预算检查
    input_tokens = 0
    if budget:
        input_tokens = budget.before_call(messages)

    response = await client.chat.completions.create(
        model=model,
        messages=messages,
        tools=tools,
        timeout=30.0,
    )

    # 预算累加
    if budget and response.usage:
        budget.after_call(
            response.usage.prompt_tokens or input_tokens,
            response.usage.completion_tokens or 0
        )

    msg = response.choices[0].message

    # FC工具循环：模型可能多次调用工具
    tool_rounds = 0
    while msg.tool_calls and tool_rounds < max_tool_rounds:
        tool_rounds += 1

        # 追加assistant消息
        messages.append({
            "role": "assistant",
            "content": msg.content,
            "tool_calls": [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {"name": tc.function.name, "arguments": tc.function.arguments}
                }
                for tc in msg.tool_calls
            ]
        })

        # 执行每个工具调用
        for tc in msg.tool_calls:
            import json
            args = json.loads(tc.function.arguments)
            result = await execute_tool(tc.function.name, args, mode="build")
            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": result
            })

        # 继续对话
        if budget:
            budget.before_call(messages)  # 检查预算
        response = await client.chat.completions.create(
            model=model,
            messages=messages,
            tools=tools,
            timeout=30.0,
        )
        if budget and response.usage:
            budget.after_call(
                response.usage.prompt_tokens or 0,
                response.usage.completion_tokens or 0
            )
        msg = response.choices[0].message

    return msg.content or ""
```

```python
async def init_db():
```
